namespace = ha_ig

##############################################################################
# ha_ig.001 — Control Panel
# Opened by the house_ambition_iron_grip decision.
# Enable/disable re-fire the same event to refresh the display.
##############################################################################
ha_ig.001 = {
	type = character_event
	title = ha_ig.001.t
	theme = realm
	left_portrait = { character = root }

	desc = {
		triggered_desc = {
			trigger = { any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty } }
			desc = ha_ig.001.status_dynasty
		}
		triggered_desc = {
			trigger = {
				NOT = { any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty } }
				any_in_global_list = { variable = ha_ig_house_list  this = root.house }
			}
			desc = ha_ig.001.status_house
		}
		triggered_desc = {
			trigger = {
				NOT = { any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty } }
				NOT = { any_in_global_list = { variable = ha_ig_house_list  this = root.house } }
			}
			desc = ha_ig.001.status_off
		}
	}

	# Whole dynasty (default). Every house of the dynasty competes.
	option = {
		name = ha_ig.001.opt.dynasty
		trigger = { NOT = { any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty } } }
		remove_list_global_variable = { name = ha_ig_house_list  target = root.house }
		add_to_global_variable_list = { name = ha_ig_dynasty_list  target = root.dynasty }
		debug_log = "iron_grip: panel - DYNASTY mode"
		trigger_event = { id = ha_ig.001 }
	}

	# This house only.
	option = {
		name = ha_ig.001.opt.house
		trigger = { NOT = { any_in_global_list = { variable = ha_ig_house_list  this = root.house } } }
		remove_list_global_variable = { name = ha_ig_dynasty_list  target = root.dynasty }
		add_to_global_variable_list = { name = ha_ig_house_list  target = root.house }
		debug_log = "iron_grip: panel - HOUSE mode"
		trigger_event = { id = ha_ig.001 }
	}

	# Turn it off for everyone.
	option = {
		name = ha_ig.001.opt.disable
		trigger = {
			OR = {
				any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty }
				any_in_global_list = { variable = ha_ig_house_list  this = root.house }
			}
		}
		remove_list_global_variable = { name = ha_ig_dynasty_list  target = root.dynasty }
		remove_list_global_variable = { name = ha_ig_house_list  target = root.house }
		debug_log = "iron_grip: panel - DISABLED"
		trigger_event = { id = ha_ig.001 }
	}

	option = {
		name = ha_ig.001.opt.close
	}
}

##############################################################################
# ha_ig.100 — Monthly tick (CK3 has no monthly pulse, so we self-reschedule)
# root = the human player. Reschedules itself every 30 days, then runs the
# sweep when this player's house has Iron Grip enabled. Seeded from the
# on_action file; on the player's death the pending event is dropped and the
# quarterly pulse re-seeds the new player.
##############################################################################
ha_ig.100 = {
	hidden = yes
	type = character_event

	immediate = {
		trigger_event = { id = ha_ig.100  days = 30 }
		if = {
			limit = {
				OR = {
					any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty }
					AND = {
						exists = house
						any_in_global_list = { variable = ha_ig_house_list  this = root.house }
					}
				}
			}
			ha_ig_sweep_effect = yes
		}
	}
}
