###############################################################################
# House Ambition: Iron Grip — on_action
#
# CK3 has NO monthly playable pulse, so the monthly cadence is driven by the
# self-rescheduling event ha_ig.100. ha_ig_player_setup_effect starts that
# chain on the human player (once per character) and applies the default mode
# (whole dynasty) the first time the dynasty is seen. We seed it:
#   - at game start (immediate on a fresh game), and
#   - on the quarterly pulse as a safety net (succession, or the mod added to
#     an existing save).
###############################################################################

on_game_start_after_lobby = {
	on_actions = {
		ha_ig_seed_monthly_on_start
	}
}

ha_ig_seed_monthly_on_start = {
	effect = {
		every_player = {
			limit = { is_ai = no }
			ha_ig_player_setup_effect = yes
		}
	}
}

quarterly_playable_pulse = {
	on_actions = {
		ha_ig_seed_monthly_tick
	}
}

ha_ig_seed_monthly_tick = {
	trigger = {
		is_ai = no
	}
	effect = {
		ha_ig_player_setup_effect = yes
	}
}
