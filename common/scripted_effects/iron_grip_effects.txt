###############################################################################
# House Ambition: Iron Grip — scripted effects
###############################################################################

# Entry point. root = player (the monthly clock). Activates the AI rulers:
#   - Dynasty mode (dynasty is in ha_ig_dynasty_list): every AI dynasty member,
#     so all houses of the dynasty compete — each still only targets vassals
#     outside its OWN house, so cadet houses act against the player's house too.
#   - House mode (house is in ha_ig_house_list): only AI members of root's house.
# Each activated ruler runs the three passes relative to their own house.
ha_ig_sweep_effect = {
	debug_log = "iron_grip: ===== starting sweep ====="
	if = {
		limit = {
			any_in_global_list = { variable = ha_ig_dynasty_list  this = root.dynasty }
		}
		debug_log = "iron_grip: dynasty mode - activating all AI dynasty rulers"
		root.dynasty = {
			every_dynasty_member = {
				limit = {
					is_ai = yes
					is_landed = yes
					NOT = { is_imprisoned = yes }
					NOT = { has_trait = incapable }
				}
				ha_ig_house_law_effect = yes
			}
		}
	}
	else_if = {
		limit = {
			any_in_global_list = { variable = ha_ig_house_list  this = root.house }
		}
		debug_log = "iron_grip: house mode - activating AI rulers of root's house"
		root.house = {
			every_house_member = {
				limit = {
					is_ai = yes
					is_landed = yes
					NOT = { is_imprisoned = yes }
					NOT = { has_trait = incapable }
				}
				ha_ig_house_law_effect = yes
			}
		}
	}
	else = {
		debug_log = "iron_grip: sweep called but neither dynasty nor house mode is enabled"
	}
	debug_log = "iron_grip: ===== sweep finished ====="
}

# Per-player setup, called from the on_action seeders. Starts this player's
# monthly tick chain (once, guarded by a character flag) and applies the
# default mode the first time this dynasty is ever seen.
ha_ig_player_setup_effect = {
	if = {
		limit = { NOT = { has_character_flag = ha_ig_monthly_tick_running } }
		add_character_flag = ha_ig_monthly_tick_running
		trigger_event = { id = ha_ig.100 }
	}
	ha_ig_apply_default_effect = yes
}

# Default mode = whole dynasty. Applied once per dynasty (guarded by a dynasty
# variable) so it never overrides a choice the player later makes in the panel.
ha_ig_apply_default_effect = {
	if = {
		limit = { dynasty = { NOT = { has_variable = ha_ig_initialized } } }
		dynasty = { set_variable = ha_ig_initialized }
		add_to_global_variable_list = {
			name = ha_ig_dynasty_list
			target = dynasty
		}
		debug_log = "iron_grip: default applied - dynasty mode enabled for this dynasty"
	}
}

# Per-ruler dispatcher. this = the AI house member ruler.
# Imprison & revoke are aggressive and can trigger revolts, so they pause
# entirely while the ruler is in ANY war. Grant (giving land to own house) is
# harmless and always runs.
ha_ig_house_law_effect = {
	save_scope_as = ha_ig_ruler
	house = { save_scope_as = ha_ig_house }
	debug_log = "iron_grip: ----- running passes for ruler below (ha_ig_ruler) -----"
	debug_log_scopes = yes
	if = {
		limit = { is_at_war = no }
		ha_ig_imprison_effect = yes
		ha_ig_revoke_effect = yes
	}
	else = {
		debug_log = "iron_grip: ruler is at war - imprison & revoke paused until peace"
	}
	ha_ig_grant_effect = yes
}

# Pass 1 — Imprison via the REAL vanilla interaction, ONE criminal per sweep so
# the ruler never starts several revolts at once. run_interaction sends the
# actual imprison_interaction (send_threshold = decline = send even if refusal
# is expected); vanilla's success-chance roll, on_accept (jail) and on_decline
# (rebellion / flee) then resolve exactly as in the base game. If the target
# revolts the ruler is now at war, and the dispatcher's war gate pauses this
# pass until peace. Skipped while the ruler is on the post-imprison hostile-
# action cooldown (vanilla's 10-day flag_hostile_actions_disabled_delay), which
# is what blocks a second send in the same window.
# Per-vassal gates: non-house, not already imprisoned, ruler has an imprisonment
# reason, ruler militarily stronger, interaction currently valid.
ha_ig_imprison_effect = {
	if = {
		limit = { scope:ha_ig_ruler = { has_character_flag = flag_hostile_actions_disabled_delay } }
		debug_log = "iron_grip: pass 1 - ruler on hostile-action cooldown, skipping this sweep"
	}
	else = {
		random_vassal = {
			limit = {
				NOT = { house = scope:ha_ig_house }
				is_imprisoned = no
				scope:ha_ig_ruler = { has_imprisonment_reason = prev }
				scope:ha_ig_ruler = { max_military_strength > prev.max_military_strength }
				scope:ha_ig_ruler = {
					is_character_interaction_valid = {
						recipient = prev
						interaction = imprison_interaction
					}
				}
			}
			save_scope_as = ha_ig_vassal
			debug_log = "iron_grip: pass 1 - RUNNING vanilla imprison_interaction (scopes below)"
			debug_log_scopes = yes
			scope:ha_ig_ruler = {
				run_interaction = {
					interaction = imprison_interaction
					actor = scope:ha_ig_ruler
					recipient = scope:ha_ig_vassal
					send_threshold = decline
				}
			}
		}
	}
}

# Pass 2 — Revoke. Reason checks evaluated FROM the ruler.
ha_ig_revoke_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 2 - ruler has no direct vassals"
	}
	else = {
		every_vassal = {
			save_scope_as = ha_ig_vassal
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 2 - skip vassal: same house"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { has_revoke_title_reason = scope:ha_ig_vassal }
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: ruler has no revoke reason"
			}
			else_if = {
				limit = {
					NOT = {
						any_held_title = {
							is_landless_type_title = no
							tier >= tier_county
						}
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: no revokable landed title (county+)"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { max_military_strength > scope:ha_ig_vassal.max_military_strength }
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: not weaker than ruler"
			}
			else = {
				debug_log = "iron_grip: pass 2 - REVOKING primary title (ruler/vassal scopes below)"
				debug_log_scopes = yes
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = ha_ig_change
					add_claim_on_loss = no
				}
				primary_title = {
					change_title_holder_include_vassals = {
						holder = scope:ha_ig_ruler
						change = scope:ha_ig_change
					}
				}
				resolve_title_and_vassal_change = scope:ha_ig_change
				scope:ha_ig_ruler = {
					consume_revoke_title_reason = scope:ha_ig_vassal
				}
			}
		}
	}
}

# Pass 3 — Grant. Loops until within domain limit or no recipient/county left.
ha_ig_grant_effect = {
	if = {
		limit = { domain_size > domain_limit }
		debug_log = "iron_grip: pass 3 - ruler over domain limit, starting grant loop"
		while = {
			limit = {
				domain_size > domain_limit
				NOT = { has_local_variable = ha_ig_grant_stop }
			}
			ha_ig_grant_one_effect = yes
		}
		remove_local_variable = ha_ig_grant_stop
		debug_log = "iron_grip: pass 3 - grant loop ended"
	}
	else = {
		debug_log = "iron_grip: pass 3 - ruler within domain limit, skipped"
	}
}

# One grant iteration. Rebuilds candidates each time (new grantees become landed).
ha_ig_grant_one_effect = {
	debug_log = "iron_grip: pass 3 - grant iteration, scanning house members"
	every_in_list = {
		list = ha_ig_grant_candidates
		remove_from_list = ha_ig_grant_candidates
	}
	clear_saved_scope = ha_ig_recipient
	clear_saved_scope = ha_ig_title

	scope:ha_ig_house = {
		every_house_member = {
			if = {
				limit = { this = scope:ha_ig_ruler }
				debug_log = "iron_grip: pass 3 - skip member: is the ruler"
			}
			else_if = {
				limit = { is_landed = yes }
				debug_log = "iron_grip: pass 3 - skip member: already landed"
			}
			else_if = {
				limit = { NOT = { is_adult = yes } }
				debug_log = "iron_grip: pass 3 - skip member: not adult"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 3 - skip member: imprisoned"
			}
			else_if = {
				limit = { has_trait = incapable }
				debug_log = "iron_grip: pass 3 - skip member: incapable"
			}
			else_if = {
				limit = {
					is_female = yes
					is_married = yes
					NOT = { matrilinear_marriage = yes }
					any_spouse = {
						is_male = yes
						patrilinear_marriage = yes
						NOT = { house = scope:ha_ig_house }
					}
				}
				debug_log = "iron_grip: pass 3 - skip member: outsider patrilineal marriage"
			}
			else_if = {
				limit = {
					NOT = {
						OR = {
							primary_heir ?= { house = scope:ha_ig_house }
							AND = {
								is_male = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								matrilinear_marriage = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								NOT = { is_married = yes }
								fertility >= 0.1
							}
						}
					}
				}
				debug_log = "iron_grip: pass 3 - skip member: fails heir-safety criteria"
			}
			else = {
				debug_log = "iron_grip: pass 3 - candidate OK, added to list"
				add_to_list = ha_ig_grant_candidates
			}
		}
	}

	random_in_list = {
		list = ha_ig_grant_candidates
		weight = {
			base = 10
			modifier = {
				add = 20
				is_courtier_of = scope:ha_ig_ruler
			}
			modifier = {
				add = 30
				primary_heir ?= { house = scope:ha_ig_house }
			}
		}
		save_scope_as = ha_ig_recipient
	}

	if = {
		limit = { exists = scope:ha_ig_recipient }
		random_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = scope:ha_ig_ruler.primary_title }
				NOT = { this = scope:ha_ig_ruler.capital_county }
			}
			save_scope_as = ha_ig_title
		}
		if = {
			limit = { exists = scope:ha_ig_title }
			debug_log = "iron_grip: pass 3 - GRANTING county (recipient/title scopes below)"
			debug_log_scopes = yes
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = ha_ig_grant_change
				add_claim_on_loss = no
			}
			scope:ha_ig_title = {
				change_title_holder = {
					holder = scope:ha_ig_recipient
					change = scope:ha_ig_grant_change
				}
			}
			resolve_title_and_vassal_change = scope:ha_ig_grant_change
		}
		else = {
			debug_log = "iron_grip: pass 3 - STOP: no grantable county (all primary/capital)"
			set_local_variable = {
				name = ha_ig_grant_stop
				value = yes
			}
		}
	}
	else = {
		debug_log = "iron_grip: pass 3 - STOP: no eligible recipient"
		set_local_variable = {
			name = ha_ig_grant_stop
			value = yes
		}
	}
}
