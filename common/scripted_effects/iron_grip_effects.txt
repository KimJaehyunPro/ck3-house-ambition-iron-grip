###############################################################################
# House Ambition: Iron Grip — scripted effects
###############################################################################

# ---------------------------------------------------------------------------
# Entry point called by the monthly pulse and the debug decision.
# root = the player character (monthly_playable_pulse always fires for the
# player).  We navigate to root.house and iterate every AI house member from
# there; each AI member is saved as scope:ha_ig_ruler inside the dispatcher.
# ---------------------------------------------------------------------------
ha_ig_sweep_effect = {
	debug_log = "iron_grip: starting monthly sweep"
	root.house = {
		every_house_member = {
			limit = {
				is_ai = yes
				is_landed = yes
				NOT = { is_imprisoned = yes }
				NOT = { has_trait = incapable }
			}
			debug_log = "iron_grip: running passes for AI house-member ruler"
			ha_ig_house_law_effect = yes
		}
	}
}

# ---------------------------------------------------------------------------
# Per-ruler dispatcher.  Called once per AI ruler from ha_ig_sweep_effect.
# this = the AI house member ruler throughout.
# scope:ha_ig_ruler — the AI ruler; used in every pass instead of root,
#   because root is always the player and never changes.
# scope:ha_ig_house  — the shared house object.
# ---------------------------------------------------------------------------
ha_ig_house_law_effect = {
	save_scope_as = ha_ig_ruler
	house = { save_scope_as = ha_ig_house }
	debug_log = "iron_grip: effect running"
	ha_ig_imprison_effect = yes
	ha_ig_revoke_effect = yes
	ha_ig_grant_effect = yes
}

# ---------------------------------------------------------------------------
# Pass 1 — Imprison
# For every direct vassal, logs exactly which condition rejected them.
# ---------------------------------------------------------------------------
ha_ig_imprison_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 1 - ruler has no direct vassals at all"
	}
	else = {
		debug_log = "iron_grip: pass 1 - scanning direct vassals"
		every_vassal = {
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 1 - skip: vassal is in the same house"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 1 - skip: vassal is already imprisoned"
			}
			else_if = {
				limit = { NOT = { has_imprisonment_reason = scope:ha_ig_ruler } }
				debug_log = "iron_grip: pass 1 - skip: ruler has no imprisonment reason against this vassal"
			}
			else_if = {
				limit = {
					NOT = { scope:ha_ig_ruler = { max_military_strength > prev.max_military_strength } }
				}
				debug_log = "iron_grip: pass 1 - skip: vassal has equal or greater max military strength than ruler"
			}
			else = {
				save_scope_as = ha_ig_prisoner
				debug_log = "iron_grip: pass 1 - ALL CONDITIONS MET - imprisoning vassal"
				scope:ha_ig_ruler = {
					imprison = {
						target = scope:ha_ig_prisoner
						type = dungeon
					}
				}
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Pass 2 — Revoke
# For every direct vassal, logs exactly which condition rejected them.
# ---------------------------------------------------------------------------
ha_ig_revoke_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 2 - ruler has no direct vassals at all"
	}
	else = {
		debug_log = "iron_grip: pass 2 - scanning direct vassals"
		every_vassal = {
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 2 - skip: vassal is in the same house"
			}
			else_if = {
				limit = { NOT = { has_revoke_title_reason = scope:ha_ig_ruler } }
				debug_log = "iron_grip: pass 2 - skip: ruler has no revoke reason against this vassal"
			}
			else_if = {
				limit = {
					NOT = {
						any_held_title = {
							is_landless_type_title = no
							tier >= tier_county
						}
					}
				}
				debug_log = "iron_grip: pass 2 - skip: vassal holds no revokable landed title (county+)"
			}
			else_if = {
				limit = {
					NOT = { scope:ha_ig_ruler = { max_military_strength > prev.max_military_strength } }
				}
				debug_log = "iron_grip: pass 2 - skip: vassal has equal or greater max military strength than ruler"
			}
			else = {
				save_scope_as = ha_ig_revoke_target
				debug_log = "iron_grip: pass 2 - ALL CONDITIONS MET - revoking from vassal"

				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = ha_ig_change
					add_claim_on_loss = no
				}
				primary_title = {
					change_title_holder_include_vassals = {
						holder = scope:ha_ig_ruler
						change = scope:ha_ig_change
					}
				}
				resolve_title_and_vassal_change = scope:ha_ig_change
				consume_revoke_title_reason = scope:ha_ig_ruler
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Pass 3 — Grant
# Only fires when scope:ha_ig_ruler is over their own domain limit.
# Iterates every_house_member on scope:ha_ig_house; the diagnostic loop
# both logs the rejection reason for each candidate AND builds the list in
# its else-branch, so no second loop is needed.
#
# Safe recipient criteria:
#   A) Has a primary heir already in the house
#   B) Male: has a safe child OR fertility >= 0.1
#   C) Female, matrilineal: has a safe child OR fertility >= 0.1
#   D) Female, unmarried: fertility >= 0.1
#
# Excluded regardless: imprisoned, incapable, landed, not adult, female in
#   a patrilineal marriage to an outsider.
# ---------------------------------------------------------------------------
ha_ig_grant_effect = {
	if = {
		limit = { domain_size > domain_limit }

		debug_log = "iron_grip: pass 3 - ruler is over domain limit, scanning house members"

		# Clear any stale entries left by the previous house member's call.
		every_in_list = {
			list = ha_ig_grant_candidates
			remove_from_list = ha_ig_grant_candidates
		}

		# Diagnostic loop: logs rejection reason for each member, adds passing
		# ones to ha_ig_grant_candidates in the else-branch.
		scope:ha_ig_house = {
			every_house_member = {
				if = {
					limit = { this = scope:ha_ig_ruler }
					debug_log = "iron_grip: pass 3 - skip: this is the ruler themselves"
				}
				else_if = {
					limit = { is_landed = yes }
					debug_log = "iron_grip: pass 3 - skip: house member is landed"
				}
				else_if = {
					limit = { NOT = { is_adult = yes } }
					debug_log = "iron_grip: pass 3 - skip: house member is not yet an adult"
				}
				else_if = {
					limit = { is_imprisoned = yes }
					debug_log = "iron_grip: pass 3 - skip: house member is imprisoned"
				}
				else_if = {
					limit = { has_trait = incapable }
					debug_log = "iron_grip: pass 3 - skip: house member has incapable trait"
				}
				else_if = {
					limit = {
						is_female = yes
						is_married = yes
						NOT = { matrilinear_marriage = yes }
						any_spouse = {
							is_male = yes
							patrilinear_marriage = yes
							NOT = { house = scope:ha_ig_house }
						}
					}
					debug_log = "iron_grip: pass 3 - skip: female in outsider patrilineal marriage (children leave house)"
				}
				else_if = {
					limit = {
						NOT = {
							OR = {
								primary_heir ?= { house = scope:ha_ig_house }
								AND = {
									is_male = yes
									OR = {
										any_child = { ha_ig_safe_child_trigger = yes }
										fertility >= 0.1
									}
								}
								AND = {
									is_female = yes
									matrilinear_marriage = yes
									OR = {
										any_child = { ha_ig_safe_child_trigger = yes }
										fertility >= 0.1
									}
								}
								AND = {
									is_female = yes
									NOT = { is_married = yes }
									fertility >= 0.1
								}
							}
						}
					}
					debug_log = "iron_grip: pass 3 - skip: fails heir-safety criteria (fertility or child check)"
				}
				else = {
					debug_log = "iron_grip: pass 3 - CANDIDATE PASSES all checks, added to list"
					add_to_list = ha_ig_grant_candidates
				}
			}
		}

		# Report whether the list has anything.
		if = {
			limit = { NOT = { any_in_list = { list = ha_ig_grant_candidates  always = yes } } }
			debug_log = "iron_grip: pass 3 - candidate list is empty after full scan"
		}
		else = {
			debug_log = "iron_grip: pass 3 - candidate list has at least one entry, picking"
		}

		random_in_list = {
			list = ha_ig_grant_candidates
			weight = {
				base = 10
				modifier = {
					add = 20
					is_courtier_of = scope:ha_ig_ruler
				}
				modifier = {
					add = 30
					primary_heir ?= { house = scope:ha_ig_house }
				}
			}
			save_scope_as = ha_ig_recipient
		}

		if = {
			limit = { exists = scope:ha_ig_recipient }
			debug_log = "iron_grip: pass 3 - recipient picked, searching for a grantable county"

			random_held_title = {
				limit = {
					tier = tier_county
					NOT = { this = scope:ha_ig_ruler.primary_title }
					NOT = { this = scope:ha_ig_ruler.capital_county }
				}
				save_scope_as = ha_ig_title
			}

			if = {
				limit = { exists = scope:ha_ig_title }
				debug_log = "iron_grip: pass 3 - ALL CONDITIONS MET - granting county to recipient"
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = ha_ig_grant_change
					add_claim_on_loss = no
				}
				scope:ha_ig_title = {
					change_title_holder = {
						holder = scope:ha_ig_recipient
						change = scope:ha_ig_grant_change
					}
				}
				resolve_title_and_vassal_change = scope:ha_ig_grant_change
			}
			else = {
				debug_log = "iron_grip: pass 3 - no grantable county found (all are primary title or capital)"
			}
		}
		else = {
			debug_log = "iron_grip: pass 3 - recipient scope not set (list was empty)"
		}
	}
	else = {
		debug_log = "iron_grip: pass 3 - ruler is within domain limit, grant pass skipped"
	}
}
