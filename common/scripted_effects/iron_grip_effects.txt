###############################################################################
# House Ambition: Iron Grip — scripted effects
###############################################################################

# Entry point. root = player. Runs all passes for every landed AI house member.
ha_ig_sweep_effect = {
	debug_log = "iron_grip: ===== starting sweep ====="
	root.house = {
		every_house_member = {
			limit = {
				is_ai = yes
				is_landed = yes
				NOT = { is_imprisoned = yes }
				NOT = { has_trait = incapable }
			}
			ha_ig_house_law_effect = yes
		}
	}
	debug_log = "iron_grip: ===== sweep finished ====="
}

# Per-ruler dispatcher. this = the AI house member ruler.
ha_ig_house_law_effect = {
	save_scope_as = ha_ig_ruler
	house = { save_scope_as = ha_ig_house }
	debug_log = "iron_grip: ----- running passes for ruler below (ha_ig_ruler) -----"
	debug_log_scopes = yes
	ha_ig_imprison_effect = yes
	ha_ig_revoke_effect = yes
	ha_ig_grant_effect = yes
}

# Pass 1 — Imprison via the REAL vanilla interaction. run_interaction makes the
# AI ruler actually send imprison_interaction (send_threshold = decline = send
# even if the vassal will refuse), so vanilla's own success-chance roll,
# on_accept (jail) and on_decline (rebellion / flee) all resolve exactly as in
# the base game. We only gate on: non-house, has imprisonment reason (from the
# ruler), ruler militarily stronger, and the interaction being currently valid.
ha_ig_imprison_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 1 - ruler has no direct vassals"
	}
	else = {
		every_vassal = {
			save_scope_as = ha_ig_vassal
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 1 - skip vassal: same house"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 1 - skip vassal: already imprisoned"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { has_imprisonment_reason = scope:ha_ig_vassal }
					}
				}
				debug_log = "iron_grip: pass 1 - skip vassal: ruler has no imprisonment reason"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { max_military_strength > scope:ha_ig_vassal.max_military_strength }
					}
				}
				debug_log = "iron_grip: pass 1 - skip vassal: not weaker than ruler"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = {
							is_character_interaction_valid = {
								recipient = scope:ha_ig_vassal
								interaction = imprison_interaction
							}
						}
					}
				}
				debug_log = "iron_grip: pass 1 - skip vassal: imprison interaction not currently valid"
			}
			else = {
				debug_log = "iron_grip: pass 1 - RUNNING vanilla imprison_interaction (scopes below)"
				debug_log_scopes = yes
				scope:ha_ig_ruler = {
					run_interaction = {
						interaction = imprison_interaction
						actor = scope:ha_ig_ruler
						recipient = scope:ha_ig_vassal
						send_threshold = decline
					}
				}
			}
		}
	}
}

# Pass 2 — Revoke. Reason checks evaluated FROM the ruler.
ha_ig_revoke_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 2 - ruler has no direct vassals"
	}
	else = {
		every_vassal = {
			save_scope_as = ha_ig_vassal
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 2 - skip vassal: same house"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { has_revoke_title_reason = scope:ha_ig_vassal }
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: ruler has no revoke reason"
			}
			else_if = {
				limit = {
					NOT = {
						any_held_title = {
							is_landless_type_title = no
							tier >= tier_county
						}
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: no revokable landed title (county+)"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { max_military_strength > scope:ha_ig_vassal.max_military_strength }
					}
				}
				debug_log = "iron_grip: pass 2 - skip vassal: not weaker than ruler"
			}
			else = {
				debug_log = "iron_grip: pass 2 - REVOKING primary title (ruler/vassal scopes below)"
				debug_log_scopes = yes
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = ha_ig_change
					add_claim_on_loss = no
				}
				primary_title = {
					change_title_holder_include_vassals = {
						holder = scope:ha_ig_ruler
						change = scope:ha_ig_change
					}
				}
				resolve_title_and_vassal_change = scope:ha_ig_change
				scope:ha_ig_ruler = {
					consume_revoke_title_reason = scope:ha_ig_vassal
				}
			}
		}
	}
}

# Pass 3 — Grant. Loops until within domain limit or no recipient/county left.
ha_ig_grant_effect = {
	if = {
		limit = { domain_size > domain_limit }
		debug_log = "iron_grip: pass 3 - ruler over domain limit, starting grant loop"
		while = {
			limit = {
				domain_size > domain_limit
				NOT = { has_local_variable = ha_ig_grant_stop }
			}
			ha_ig_grant_one_effect = yes
		}
		remove_local_variable = ha_ig_grant_stop
		debug_log = "iron_grip: pass 3 - grant loop ended"
	}
	else = {
		debug_log = "iron_grip: pass 3 - ruler within domain limit, skipped"
	}
}

# One grant iteration. Rebuilds candidates each time (new grantees become landed).
ha_ig_grant_one_effect = {
	debug_log = "iron_grip: pass 3 - grant iteration, scanning house members"
	every_in_list = {
		list = ha_ig_grant_candidates
		remove_from_list = ha_ig_grant_candidates
	}
	clear_saved_scope = ha_ig_recipient
	clear_saved_scope = ha_ig_title

	scope:ha_ig_house = {
		every_house_member = {
			if = {
				limit = { this = scope:ha_ig_ruler }
				debug_log = "iron_grip: pass 3 - skip member: is the ruler"
			}
			else_if = {
				limit = { is_landed = yes }
				debug_log = "iron_grip: pass 3 - skip member: already landed"
			}
			else_if = {
				limit = { NOT = { is_adult = yes } }
				debug_log = "iron_grip: pass 3 - skip member: not adult"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 3 - skip member: imprisoned"
			}
			else_if = {
				limit = { has_trait = incapable }
				debug_log = "iron_grip: pass 3 - skip member: incapable"
			}
			else_if = {
				limit = {
					is_female = yes
					is_married = yes
					NOT = { matrilinear_marriage = yes }
					any_spouse = {
						is_male = yes
						patrilinear_marriage = yes
						NOT = { house = scope:ha_ig_house }
					}
				}
				debug_log = "iron_grip: pass 3 - skip member: outsider patrilineal marriage"
			}
			else_if = {
				limit = {
					NOT = {
						OR = {
							primary_heir ?= { house = scope:ha_ig_house }
							AND = {
								is_male = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								matrilinear_marriage = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								NOT = { is_married = yes }
								fertility >= 0.1
							}
						}
					}
				}
				debug_log = "iron_grip: pass 3 - skip member: fails heir-safety criteria"
			}
			else = {
				debug_log = "iron_grip: pass 3 - candidate OK, added to list"
				add_to_list = ha_ig_grant_candidates
			}
		}
	}

	random_in_list = {
		list = ha_ig_grant_candidates
		weight = {
			base = 10
			modifier = {
				add = 20
				is_courtier_of = scope:ha_ig_ruler
			}
			modifier = {
				add = 30
				primary_heir ?= { house = scope:ha_ig_house }
			}
		}
		save_scope_as = ha_ig_recipient
	}

	if = {
		limit = { exists = scope:ha_ig_recipient }
		random_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = scope:ha_ig_ruler.primary_title }
				NOT = { this = scope:ha_ig_ruler.capital_county }
			}
			save_scope_as = ha_ig_title
		}
		if = {
			limit = { exists = scope:ha_ig_title }
			debug_log = "iron_grip: pass 3 - GRANTING county (recipient/title scopes below)"
			debug_log_scopes = yes
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = ha_ig_grant_change
				add_claim_on_loss = no
			}
			scope:ha_ig_title = {
				change_title_holder = {
					holder = scope:ha_ig_recipient
					change = scope:ha_ig_grant_change
				}
			}
			resolve_title_and_vassal_change = scope:ha_ig_grant_change
		}
		else = {
			debug_log = "iron_grip: pass 3 - STOP: no grantable county (all primary/capital)"
			set_local_variable = {
				name = ha_ig_grant_stop
				value = yes
			}
		}
	}
	else = {
		debug_log = "iron_grip: pass 3 - STOP: no eligible recipient"
		set_local_variable = {
			name = ha_ig_grant_stop
			value = yes
		}
	}
}
