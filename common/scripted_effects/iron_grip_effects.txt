###############################################################################
# House Ambition: Iron Grip — scripted effects
###############################################################################

# Entry point. root = player. Runs all passes for every landed AI house member.
ha_ig_sweep_effect = {
	debug_log = "iron_grip: ===== starting sweep ====="
	root.house = {
		every_house_member = {
			limit = {
				is_ai = yes
				is_landed = yes
				NOT = { is_imprisoned = yes }
				NOT = { has_trait = incapable }
			}
			ha_ig_house_law_effect = yes
		}
	}
	debug_log = "iron_grip: ===== sweep finished ====="
}

# Per-ruler dispatcher. this = the AI house member ruler.
ha_ig_house_law_effect = {
	save_scope_as = ha_ig_ruler
	house = { save_scope_as = ha_ig_house }
	debug_log = "iron_grip: ----- running passes for ruler [ha_ig_ruler.GetName] -----"
	ha_ig_imprison_effect = yes
	ha_ig_revoke_effect = yes
	ha_ig_grant_effect = yes
}

# Pass 1 — Imprison. Reason checks evaluated FROM the ruler (direction matters).
ha_ig_imprison_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 1 - [ha_ig_ruler.GetName] has no direct vassals"
	}
	else = {
		every_vassal = {
			save_scope_as = ha_ig_vassal
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 1 - skip [ha_ig_vassal.GetName]: same house"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 1 - skip [ha_ig_vassal.GetName]: already imprisoned"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { has_imprisonment_reason = scope:ha_ig_vassal }
					}
				}
				debug_log = "iron_grip: pass 1 - skip [ha_ig_vassal.GetName]: [ha_ig_ruler.GetName] has no imprisonment reason against them"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { max_military_strength > scope:ha_ig_vassal.max_military_strength }
					}
				}
				debug_log = "iron_grip: pass 1 - skip [ha_ig_vassal.GetName]: their army ([ha_ig_vassal.MaxMilitaryStrength|0]) is not weaker than [ha_ig_ruler.GetName]'s ([ha_ig_ruler.MaxMilitaryStrength|0])"
			}
			else = {
				debug_log = "iron_grip: pass 1 - IMPRISONING [ha_ig_vassal.GetName] (by [ha_ig_ruler.GetName])"
				scope:ha_ig_ruler = {
					imprison = {
						target = scope:ha_ig_vassal
						type = dungeon
					}
				}
			}
		}
	}
}

# Pass 2 — Revoke. Reason checks evaluated FROM the ruler.
ha_ig_revoke_effect = {
	if = {
		limit = { NOT = { any_vassal = { always = yes } } }
		debug_log = "iron_grip: pass 2 - [ha_ig_ruler.GetName] has no direct vassals"
	}
	else = {
		every_vassal = {
			save_scope_as = ha_ig_vassal
			if = {
				limit = { house = scope:ha_ig_house }
				debug_log = "iron_grip: pass 2 - skip [ha_ig_vassal.GetName]: same house"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { has_revoke_title_reason = scope:ha_ig_vassal }
					}
				}
				debug_log = "iron_grip: pass 2 - skip [ha_ig_vassal.GetName]: [ha_ig_ruler.GetName] has no revoke reason against them"
			}
			else_if = {
				limit = {
					NOT = {
						any_held_title = {
							is_landless_type_title = no
							tier >= tier_county
						}
					}
				}
				debug_log = "iron_grip: pass 2 - skip [ha_ig_vassal.GetName]: no revokable landed title (county+)"
			}
			else_if = {
				limit = {
					NOT = {
						scope:ha_ig_ruler = { max_military_strength > scope:ha_ig_vassal.max_military_strength }
					}
				}
				debug_log = "iron_grip: pass 2 - skip [ha_ig_vassal.GetName]: their army ([ha_ig_vassal.MaxMilitaryStrength|0]) is not weaker than [ha_ig_ruler.GetName]'s ([ha_ig_ruler.MaxMilitaryStrength|0])"
			}
			else = {
				debug_log = "iron_grip: pass 2 - REVOKING [ha_ig_vassal.GetPrimaryTitle.GetName] from [ha_ig_vassal.GetName] (by [ha_ig_ruler.GetName])"
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = ha_ig_change
					add_claim_on_loss = no
				}
				primary_title = {
					change_title_holder_include_vassals = {
						holder = scope:ha_ig_ruler
						change = scope:ha_ig_change
					}
				}
				resolve_title_and_vassal_change = scope:ha_ig_change
				scope:ha_ig_ruler = {
					consume_revoke_title_reason = scope:ha_ig_vassal
				}
			}
		}
	}
}

# Pass 3 — Grant. Loops until within domain limit or no recipient/county left.
ha_ig_grant_effect = {
	if = {
		limit = { domain_size > domain_limit }
		debug_log = "iron_grip: pass 3 - [ha_ig_ruler.GetName] is over domain limit ([ha_ig_ruler.GetDomainSize]/[ha_ig_ruler.GetDomainLimit]), starting grant loop"
		while = {
			limit = {
				domain_size > domain_limit
				NOT = { has_local_variable = ha_ig_grant_stop }
			}
			ha_ig_grant_one_effect = yes
		}
		remove_local_variable = ha_ig_grant_stop
		debug_log = "iron_grip: pass 3 - grant loop ended for [ha_ig_ruler.GetName] ([ha_ig_ruler.GetDomainSize]/[ha_ig_ruler.GetDomainLimit])"
	}
	else = {
		debug_log = "iron_grip: pass 3 - [ha_ig_ruler.GetName] is within domain limit, skipped"
	}
}

# One grant iteration. Rebuilds candidates each time (new grantees become landed).
ha_ig_grant_one_effect = {
	every_in_list = {
		list = ha_ig_grant_candidates
		remove_from_list = ha_ig_grant_candidates
	}
	clear_saved_scope = ha_ig_recipient
	clear_saved_scope = ha_ig_title

	scope:ha_ig_house = {
		every_house_member = {
			save_scope_as = ha_ig_candidate
			if = {
				limit = { this = scope:ha_ig_ruler }
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: is the ruler"
			}
			else_if = {
				limit = { is_landed = yes }
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: already landed"
			}
			else_if = {
				limit = { NOT = { is_adult = yes } }
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: not adult"
			}
			else_if = {
				limit = { is_imprisoned = yes }
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: imprisoned"
			}
			else_if = {
				limit = { has_trait = incapable }
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: incapable"
			}
			else_if = {
				limit = {
					is_female = yes
					is_married = yes
					NOT = { matrilinear_marriage = yes }
					any_spouse = {
						is_male = yes
						patrilinear_marriage = yes
						NOT = { house = scope:ha_ig_house }
					}
				}
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: outsider patrilineal marriage"
			}
			else_if = {
				limit = {
					NOT = {
						OR = {
							primary_heir ?= { house = scope:ha_ig_house }
							AND = {
								is_male = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								matrilinear_marriage = yes
								OR = {
									any_child = { ha_ig_safe_child_trigger = yes }
									fertility >= 0.1
								}
							}
							AND = {
								is_female = yes
								NOT = { is_married = yes }
								fertility >= 0.1
							}
						}
					}
				}
				debug_log = "iron_grip: pass 3 - skip [ha_ig_candidate.GetName]: fails heir-safety criteria"
			}
			else = {
				debug_log = "iron_grip: pass 3 - candidate OK: [ha_ig_candidate.GetName]"
				add_to_list = ha_ig_grant_candidates
			}
		}
	}

	random_in_list = {
		list = ha_ig_grant_candidates
		weight = {
			base = 10
			modifier = {
				add = 20
				is_courtier_of = scope:ha_ig_ruler
			}
			modifier = {
				add = 30
				primary_heir ?= { house = scope:ha_ig_house }
			}
		}
		save_scope_as = ha_ig_recipient
	}

	if = {
		limit = { exists = scope:ha_ig_recipient }
		random_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = scope:ha_ig_ruler.primary_title }
				NOT = { this = scope:ha_ig_ruler.capital_county }
			}
			save_scope_as = ha_ig_title
		}
		if = {
			limit = { exists = scope:ha_ig_title }
			debug_log = "iron_grip: pass 3 - GRANTING [ha_ig_title.GetName] to [ha_ig_recipient.GetName] (from [ha_ig_ruler.GetName])"
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = ha_ig_grant_change
				add_claim_on_loss = no
			}
			scope:ha_ig_title = {
				change_title_holder = {
					holder = scope:ha_ig_recipient
					change = scope:ha_ig_grant_change
				}
			}
			resolve_title_and_vassal_change = scope:ha_ig_grant_change
		}
		else = {
			debug_log = "iron_grip: pass 3 - STOP: [ha_ig_ruler.GetName] has no grantable county (all primary/capital)"
			set_local_variable = {
				name = ha_ig_grant_stop
				value = yes
			}
		}
	}
	else = {
		debug_log = "iron_grip: pass 3 - STOP: no eligible recipient for [ha_ig_ruler.GetName]"
		set_local_variable = {
			name = ha_ig_grant_stop
			value = yes
		}
	}
}
